cmake_minimum_required(VERSION 3.21)
project(ds4v_image_policy LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(ds4v_image_policy STATIC ../../src/deepseek4/deepseek4_image_policy.cpp)
target_include_directories(ds4v_image_policy PUBLIC ../../src)
target_compile_options(ds4v_image_policy PRIVATE -Wall -Wextra -Werror)
add_executable(test_ds4v_image_policy test.cpp)
target_link_libraries(test_ds4v_image_policy PRIVATE ds4v_image_policy)
add_executable(ds4v_image_policy_probe probe.cpp)
target_link_libraries(ds4v_image_policy_probe PRIVATE ds4v_image_policy)
enable_testing()
add_test(NAME ds4v_image_policy COMMAND test_ds4v_image_policy)
