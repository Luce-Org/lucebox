cmake_minimum_required(VERSION 3.21)
project(ds4_bf16_affine LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(GGML_BUILD "" CACHE PATH "Immutable existing GGML build")
add_executable(ds4_bf16_affine probe.cpp)
target_include_directories(ds4_bf16_affine PRIVATE ../../src ../../deps/llama.cpp/ggml/include)
foreach(lib ggml/src/libggml-base.so.0 ggml/src/libggml-cpu.so.0 ggml/src/ggml-hip/libggml-hip.so.0)
    if(NOT EXISTS "${GGML_BUILD}/${lib}")
        message(FATAL_ERROR "Missing immutable library ${GGML_BUILD}/${lib}")
    endif()
    target_link_libraries(ds4_bf16_affine PRIVATE "${GGML_BUILD}/${lib}")
endforeach()
